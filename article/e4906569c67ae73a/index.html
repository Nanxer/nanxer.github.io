from Crypto.Util.number import bytes_to_long
from base64 import b64encode
from secret import flag
msg = flag.encode()
length = len(msg)
assert length%4 == 0
block = length//4
m = [ msg[ block*(i) : block*(i+1) ] for i in range(4) ]
m0 = m[0]
m1 = bytes_to_long(m[1])
m2 = m[2].hex()
m3 = b64encode(m[3])
print(f'm0 = {m0}\nm1 = {m1}\nm2 = {m2}\nm3 = {m3}')

from random import randint
from gmpy2 import invert,gcd
#Hash Function:
def MD5(m):return md5(str(m).encode()).hexdigest()
#RSA AlgorithmParameter Generation Function:
def KeyGen():
    Factor_BitLength = 30
    q = getPrime(Factor_BitLength)
    p = getPrime(Factor_BitLength)
    N = p * q
    #Euler's totient function:
    phi = (p-1) * (q-1)
    #Generate Keys:
    while True:
        e = randint(1,phi)
        if gcd(e,phi) == 1:
            d = int(invert(e,phi))
            break
    #Generate Result:
    Pub_Key = (N,e)
    Prv_Key = (N,d)
    return Pub_Key,Prv_Key
