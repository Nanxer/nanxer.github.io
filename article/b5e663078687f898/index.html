def jump_frog(a, b):
    while a != b:
        if a > b:
            a = a - b
        else:
            b = b - a
    return a, b

def create_inverse_mapping(cycles):
    """创建置换的逆映射字典"""
    inverse_map = {}
    for cycle in cycles:
        cycle_length = len(cycle)
        for idx, num in enumerate(cycle):

def find_p(public_keys):
    # 取前3个公钥来计算
    pk0, pk1, pk2 = public_keys[:3]

    # 尝试所有可能的偏移量组合（1-10）
    for r0 in range(1, 11):
        v0 = pk0 - r0
        if v0 <= 0:
            continue
        for r1 in range(1, 11):
            v1 = pk1 - r1
            if v1 <= 0:
                continue
            g = math.gcd(v0, v1)
            if g < 2:
                continue
            for r2 in range(1, 11):
                v2 = pk2 - r2
                if v2 <= 0:
                    continue
                current_gcd = math.gcd(g, v2)
                # p是128位素数，值会很大
                if current_gcd > 10**30:
                    return current_gcd
    return None

p = find_p(public_keys)
print(f"找到p: {p}")
